cmake_minimum_required(VERSION 3.24)
project(StarshipPadSDKProbe LANGUAGES OBJCXX)
find_library(METAL Metal REQUIRED)
find_library(OSX_FOUNDATION Foundation REQUIRED)
find_library(OSX_AVFOUNDATION AVFoundation REQUIRED)
foreach(framework METAL OSX_FOUNDATION OSX_AVFOUNDATION)
    string(FIND "${${framework}}" "${CMAKE_OSX_SYSROOT}/" sdk_prefix)
    if(NOT sdk_prefix EQUAL 0)
        message(FATAL_ERROR "${framework} escaped the selected SDK: ${${framework}}")
    endif()
endforeach()
if(NOT KEEP_SENTINEL STREQUAL "preserved")
    message(FATAL_ERROR "Unrelated cache setting was lost")
endif()
add_library(UIKitProbe OBJECT UIKitProbe.mm)
